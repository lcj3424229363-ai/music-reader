### 1.3 Score Model 当前 JSON 行为

- `class Note` / `class Score` 等都是 class instance
- `JSON.stringify(score)` 走 Object.keys, class instance 字段是 enumerable 的 (因为 `this.foo = bar` 等价于 plain property)
- 但是: **class instance 序列化后是 plain object, 反序列化时类型丢了** — `JSON.parse` 不会自动还原成 Score
- 现状: 没有 toJSON 方法, 没有 fromJSON helper

## 3. 实施范围 (按推荐决策)

### 3.1 class toJSON (5 个 class, Score 触发级联)

```js
Note.prototype.toJSON = function() {
  return {
    id: this.id, kind: this.kind, voice: this.voice,
    pitches: this.pitches ? this.pitches.map(p => ({...p})) : [],
    duration: this.duration, dotted: this.dotted, units: this.units,
    articulation: this.articulation, ...  // 全 18 修饰符 + 3 tuplet 字段
    tupletType: this.tupletType, tupletGroup: this.tupletGroup, tupletPosition: this.tupletPosition
  };
};

Relation.prototype.toJSON = function() {
  return {
    id: this.id, type: this.type,
    from: this.from, to: this.to,
    members: this.members ? [...this.members] : null,
    payload: this.payload ? {...this.payload} : {}
  };
};

Voice.prototype.toJSON = function() { return {id, role, notes: notes.map(n => n.toJSON())}; };
Measure.prototype.toJSON = function() { return {index, number, timeSignature, beginBarline, endBarline, voices: voices.map(v => v.toJSON()), tempo, directions, volta, rehearsalMark}; };
Part.prototype.toJSON = function() { return {id, name, instrument, staves, measures: measures.map(m => m.toJSON())}; };
Score.prototype.toJSON = function() { return {meta: {...meta}, parts: parts.map(p => p.toJSON()), relations: relations.map(r => r.toJSON())}; };
```

### 3.2 ScoreModel.scoreFromJSON(json) 反序列化

```js
function scoreFromJSON(json) {
  const score = new Score();
  if (json.meta) Object.assign(score.meta, json.meta);
  if (Array.isArray(json.parts)) {
    json.parts.forEach(pJson => {
      const part = new Part(pJson.id, pJson.name, pJson.instrument);
      pJson.measures.forEach(mJson => {
        const measure = new Measure(mJson.index, mJson.timeSignature);
        // ... 递归还原 voices → notes
        mJson.voices.forEach(vJson => {
          const voice = new Voice(vJson.id, vJson.role);
          vJson.notes.forEach(nJson => {
            const note = noteFromJSON(nJson);  // helper
            voice.notes.push(note);
          });
          measure.voices.push(voice);
        });
        part.measures.push(measure);
      });
      score.addPart(part);
    });
  }
  if (Array.isArray(json.relations)) {
    json.relations.forEach(rJson => {
      const rel = new Relation(rJson.type, rJson.from, rJson.to);
      rel.id = rJson.id;
      rel.members = rJson.members ? [...rJson.members] : null;
      rel.payload = rJson.payload ? {...rJson.payload} : {};
      score.addRelation(rel);
    });
  }
  return score;
}
```
