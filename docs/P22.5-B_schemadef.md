## 5 决策落地 (recap)

| # | 决策 | 落地 |
|---|------|------|
| 1 | measure 粒度 | **A per voice** — 顶层 `measures[]`, 每 measure 多个 voices |
| 2 | 跨 measure 关系 | **A reference (id)** — 独立关系对象 `{type, from, to}` |
| 3 | 节奏单位 | **A 保留 units** (1 quarter = 8) — 老字段不动 |
| 4 | editorState vs Score | **B 影子** — Score 是 computed view, 老 `scoreMeasures` 仍真 |
| 5 | VexFlow 边界 | **B 半隔离** — Renderer 唯一 VF 使用方 (P22.6 收口) |

## 数据结构 (B.1 实施目标)

```js
// P22.5-B.1 — Score Model 5 个 class + 1 个 factory
// 文件: web/score-model.js
// 加载: <script src="score-model.js"></script> (在 app.js 前)

// 1. Score (顶层) — 全局谱面
class Score {
  constructor() {
    this.meta = {           // 谱面元数据
      title: "",            // 标题
      composer: "",         // 作曲家
      tempo: 120,           // 全局默认速度 (BPM)
      anacrusis: false      // 弱起小节
    };
    this.parts = [];         // Part[] — 谱面部件 (1 个 / 大谱表 2+ 个)
    this.relations = [];     // Relation[] — 跨 measure 关系 (tie/slur/tuplet)
  }
  
  addPart(part) { ... }
  getPart(partId) { ... }
  getMeasure(partId, measureIndex) { ... }  // 直接访问
  addRelation(relation) { ... }
  getRelationsByType(type) { ... }  // type: "tie" | "slur" | "tuplet"
}

// 2. Part — 谱面部件 (大谱表 = 多个 part)
class Part {
  constructor(id, name, instrument = "piano") {
    this.id = id;                       // "treble" | "bass" | "soprano" | ...
    this.name = name;                   // "高音谱表" | "Soprano"
    this.instrument = instrument;       // "piano" | "voice" | "violin" | ...
    this.staves = [];                   // Staff[] — 通常 1 个
    this.measures = [];                 // Measure[] — 该 part 的所有小节
  }
}

// 3. Measure — 小节 (1 个 part 1 个 time slice)
class Measure {
  constructor(index, timeSignature = "4/4") {
    this.index = index;                 // 0-based 小节编号
    this.number = index + 1;           // 1-based 谱面编号
    this.timeSignature = timeSignature; // "4/4" | "3/4" | "6/8" | ...
    this.beginBarline = "single";       // "single" | "repeat-begin" | "none"
    this.endBarline = "single";         // "single" | "double" | "end" | "repeat-end" | "repeat-both"
    this.voices = [];                   // Voice[] — 决策 1 落地: per voice
    this.tempo = null;                  // 该小节单独速度 (override 全局), null 用 meta.tempo
    this.directions = [];               // Direction[] — 小节级别文本标记 (Allegro, rit., ...)
    this.volta = null;                  // 1 | 2 | 3 | null (per part, A.4 字段)
    this.rehearsalMark = "";            // "A" / "Intro" / "Verse 1" (per part, A.4 字段)
  }
}

// 4. Voice — 声部 (决策 1 落地)
class Voice {
  constructor(id, role = "primary") {
    this.id = id;                       // "1" | "2" — 跟现有 voiceId 兼容
    this.role = role;                   // "primary" | "secondary" | "bass" | ...
    this.entries = [];                  // Entry[] — 该声部的 entry
  }
}

// 5. Entry — 单个音符/休止符
class Entry {
  constructor(kind = "note") {
    this.id = "";                       // 唯一 id (B.2 生成: "p0m0v0e0")
    this.kind = kind;                   // "note" | "rest" | "placeholder" | "chord" | "grace"
    this.voice = "1";                   // "1" | "2" — 冗余 (voice.id 已存), 兼容老代码
    this.pitches = [];                  // Pitch[] (kind=note) | [] (kind=rest)
    this.duration = "q";                // "w" | "h" | "q" | "8" | "16" | ...
    this.dotted = false;                // 附点
    this.units = 8;                     // 1 quarter = 8 units (决策 3 落地)
    
    // 18 修饰符字段 (P22.5-Symbol A.4 落地)
    this.articulation = "";
    this.fingering = "";
    this.textMark = "";
    this.ornament = "";
    this.fermata = false;
    this.grace = null;
    this.dynamic = "";
    this.pedal = "";
    this.breath = false;
    this.tieStart = false;
    this.tieStop = false;
    this.slurStart = false;
    this.slurStop = false;
    this.phraseStart = false;
    this.phraseStop = false;
    this.hairpin = "";
    this.chordSymbol = "";
    this.arpeggiate = false;
    // 曲式 (P22.5-Symbol A.4)
    this.volta = null;
    this.rehearsalMark = "";
  }
}

// 6. Relation — 跨 measure 关系 (决策 2 落地)
class Relation {
  constructor(type, fromId, toId) {
    this.id = "";                       // "r0", "r1", ...
    this.type = type;                   // "tie" | "slur" | "phrase" | "tuplet"
    this.from = fromId;                 // Entry.id
    this.to = toId;                     // Entry.id
    this.payload = {};                  // 类型特定: tuplet.ratio = "3:2", slur.style = "solid" 等
  }
}
```

## 与现状的映射 (B.2 转换层)

| 现状 | Score Model | 备注 |
|------|-------------|------|
| `scoreMeasures[i]` | `score.parts[0].measures[i]` | 单谱表 |
| `staffScores.treble.measures[i]` | `score.parts[0].measures[i]` | 大谱表 treble |
| `staffScores.bass.measures[i]` | `score.parts[1].measures[i]` | 大谱表 bass |
| `entry.tieStart: true` | `entry.tieStart: true` + `score.relations.push(Relation("tie", id, targetId))` | 加 id + 关系对象 |
| `entry.tupletType: "triplet"` | `entry.tupletType: "triplet"` + `score.relations` | 老字段保留, 关系 B.3 补 |
| `entry.voice: "1"` | `entry.voice: "1"` + `score.parts[i].voices[j].entries[k]` | 冗余, 兼容 |
| `editorState` | **不动** | 决策 4 影子模式 |
| `VF.Articulation(...)` | **不动** (P22.6 改) | 决策 5 Renderer 隔离 |
